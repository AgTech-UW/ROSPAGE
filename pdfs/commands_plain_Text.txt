==============================================================================
PART 6: PLAN B -- NATIVE WSL INSTALL
==============================================================================
Only do this if Part 5 won't cooperate.


# --- Step 1: Install Ubuntu in WSL (skip if already installed) ---
wsl --install -d Ubuntu-22.04


# --- Step 2: Install ROS 2 Humble (run inside WSL Ubuntu) ---

# Set locale
sudo apt update && sudo apt install -y locales
sudo locale-gen en_US en_US.UTF-8
sudo update-locale LC_ALL=en_US.UTF-8 LANG=en_US.UTF-8
export LANG=en_US.UTF-8

# Add ROS 2 apt repository
sudo apt install -y software-properties-common curl gnupg lsb-release
sudo add-apt-repository universe
sudo curl -sSL https://raw.githubusercontent.com/ros/rosdistro/master/ros.key \
    -o /usr/share/keyrings/ros-archive-keyring.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/ros-archive-keyring.gpg] http://packages.ros.org/ros2/ubuntu $(. /etc/os-release && echo $UBUNTU_CODENAME) main" \
    | sudo tee /etc/apt/sources.list.d/ros2.list > /dev/null

# Install ROS Humble + dev tools + Cyclone
sudo apt update
sudo apt install -y ros-humble-desktop python3-colcon-common-extensions \
    python3-rosdep python3-vcstool python3-pip \
    ros-humble-rmw-cyclonedds-cpp

# Source ROS in every new shell (CHANGE THE 5 to your domain ID)
echo 'source /opt/ros/humble/setup.bash' >> ~/.bashrc
echo 'export RMW_IMPLEMENTATION=rmw_cyclonedds_cpp' >> ~/.bashrc
echo 'export ROS_DOMAIN_ID=5' >> ~/.bashrc
source ~/.bashrc

# Initialize rosdep
sudo rosdep init || true
rosdep update


# --- Step 3: Lab dependencies ---
pip install pynput numpy matplotlib


# --- Step 4: Move your code into WSL ---

# OPTION A: clone from Git (recommended)
cd ~
mkdir -p ros2_ws/src
cd ros2_ws/src
git clone <your_repo_url> agtech_ros2

# OPTION B: copy out from old dev container first, then into WSL
# Inside the old dev container:
cp -r /workspaces/agtech_ros2 /mnt/c/Users/<you>/Desktop/agtech_backup
# Then in WSL (NOT in the container):
mkdir -p ~/ros2_ws/src
cp -r /mnt/c/Users/<you>/Desktop/agtech_backup ~/ros2_ws/src/agtech_ros2


# --- Step 5: Symlink so /workspaces/agtech_ros2 still works ---
sudo mkdir -p /workspaces
sudo ln -s ~/ros2_ws/src/agtech_ros2 /workspaces/agtech_ros2


# --- Step 6: Build ---
cd ~/ros2_ws
colcon build --symlink-install
echo 'source ~/ros2_ws/install/setup.bash' >> ~/.bashrc
source ~/.bashrc


# --- Step 7: Open VS Code with the WSL extension ---
# Install the "WSL" extension in VS Code first (not "Dev Containers"!)
code ~/ros2_ws


# --- Step 8: Verify ---
pkill -f ros2_daemon
ros2 topic list
